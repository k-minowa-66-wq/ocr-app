function doPost(e) { try { if (!e || !e.postData || !e.postData.contents) { return createJsonResponse({ success: false, error: "送信データ(postData)を受信できませんでした。" }); } var requestData = JSON.parse(e.postData.contents); var action = requestData.action; if (action === "callGeminiAPI") { var prompt = requestData.prompt; var maxTokens = requestData.maxTokens || 1500; var apiResponse = fetchGeminiWithDynamicFallback(prompt, maxTokens); return createJsonResponse({ success: true, response: apiResponse }); } if (action === "callVisionOCR") { var base64ImageData = requestData.base64ImageData; var ocrResultText = processVisionOCRWithFallback(base64ImageData); return createJsonResponse({ success: true, response: ocrResultText }); } if (action === "saveLearningLog") { saveLogToSpreadsheet(requestData.logData); return createJsonResponse({ success: true, message: "ログの保存に成功しました。" }); } if (action === "getLearningHistory") { var logs = getHistoryFromSpreadsheet(requestData.userEmail); return createJsonResponse({ success: true, logs: logs }); } return createJsonResponse({ success: false, error: "未知のアクション: " + action }); } catch (err) { return createJsonResponse({ success: false, error: err.toString() }); } } /** * 手書き画像文字起こし（動的アクティブモデル参照 ＋ 万全のフォールバック） */ function processVisionOCRWithFallback(base64ImageData) { var props = PropertiesService.getScriptProperties(); var geminiKey = props.getProperty("GEMINI_API_KEY") || props.getProperty("GEMINI_API_KEY_1"); var groqKey = props.getProperty("GROQ_API_KEY"); if (!geminiKey && !groqKey) { throw new Error("スクリプトプロパティに GEMINI_API_KEY または GROQ_API_KEY が設定されていません。"); } var ocrPrompt = "Transcribe the handwritten English essay image exact text as it is. Do NOT correct grammar, spelling, or punctuation. Output ONLY the transcribed English text, nothing else."; var mimeType = "image/jpeg"; var base64Clean = base64ImageData; if (base64ImageData.indexOf(",") !== -1) { var parts = base64ImageData.split(","); var match = parts[0].match(/data:(.*?);base64/); if (match) mimeType = match[1]; base64Clean = parts[1]; } var errorDetails = []; // 1. Gemini Vision（動的取得したアクティブモデルを順次試行） if (geminiKey) { var activeModels = getActiveGeminiModels(geminiKey); for (var i = 0; i < activeModels.length; i++) { try { var currentModel = activeModels[i]; var url = "https://generativelanguage.googleapis.com/v1beta/" + currentModel + ":generateContent?key=" + geminiKey; var payload = { contents: [{ parts: [ { text: ocrPrompt }, { inline_data: { mime_type: mimeType, data: base64Clean } } ] }], generationConfig: { temperature: 0.1, maxOutputTokens: 1000 } }; var res = UrlFetchApp.fetch(url, { method: "post", contentType: "application/json", payload: JSON.stringify(payload), muteHttpExceptions: true }); var code = res.getResponseCode(); var jsonText = res.getContentText(); if (code === 200) { var json = JSON.parse(jsonText); if (json.candidates && json.candidates[0] && json.candidates[0].content && json.candidates[0].content.parts[0]) { return json.candidates[0].content.parts[0].text.trim(); } } else { errorDetails.push("Gemini(" + currentModel + ") Error " + code + ": " + jsonText.substring(0, 120)); } } catch (e) { errorDetails.push("Gemini通信例外: " + e.toString()); } } } // 2. Groq Vision（現行稼働中のVisionモデルへ切替試行） if (groqKey) { var groqModels = ["llama-3.2-11b-vision-instruct", "llama-3.2-90b-vision-instruct", "meta-llama/llama-4-scout-17b-16e-instruct"]; for (var j = 0; j < groqModels.length; j++) { try { var groqUrl = "https://api.groq.com/openai/v1/chat/completions"; var groqPayload = { model: groqModels[j], messages: [{ role: "user", content: [ { type: "text", text: ocrPrompt }, { type: "image_url", image_url: { url: "data:" + mimeType + ";base64," + base64Clean } } ] }], temperature: 0.1, max_tokens: 1000 }; var groqRes = UrlFetchApp.fetch(groqUrl, { method: "post", headers: { "Authorization": "Bearer " + groqKey }, contentType: "application/json", payload: JSON.stringify(groqPayload), muteHttpExceptions: true }); var gCode = groqRes.getResponseCode(); var gText = groqRes.getContentText(); if (gCode === 200) { var groqJson = JSON.parse(gText); if (groqJson.choices && groqJson.choices[0] && groqJson.choices[0].message) { return groqJson.choices[0].message.content.trim(); } } else { errorDetails.push("Groq(" + groqModels[j] + ") Error " + gCode + ": " + gText.substring(0, 120)); } } catch (e) { errorDetails.push("Groq通信例外: " + e.toString()); } } } throw new Error("⚠️ 利用可能な全AIモデルで文字起こしに失敗しました。\n詳細:\n" + errorDetails.join("\n")); } /** * Google AIから現在アクティブなモデル一覧を動的に取得（6時間キャッシュ） */ function getActiveGeminiModels(apiKey) { var cache = CacheService.getScriptCache(); var cached = cache.get("ACTIVE_GEMINI_MODELS"); if (cached) { try { return JSON.parse(cached); } catch (e) {} } try { var listUrl = "https://generativelanguage.googleapis.com/v1beta/models?key=" + apiKey; var res = UrlFetchApp.fetch(listUrl, { method: "get", muteHttpExceptions: true }); if (res.getResponseCode() === 200) { var data = JSON.parse(res.getContentText()); if (data.models && data.models.length > 0) { var validModels = data.models.filter(function(m) { return m.supportedGenerationMethods && m.supportedGenerationMethods.indexOf("generateContent") !== -1; }); validModels.sort(function(a, b) { var aFlash = a.name.indexOf("flash") !== -1 ? 1 : 0; var bFlash = b.name.indexOf("flash") !== -1 ? 1 : 0; if (aFlash !== bFlash) return bFlash - aFlash; return b.name.localeCompare(a.name); }); var modelNames = validModels.map(function(m) { return m.name; }); if (modelNames.length > 0) { cache.put("ACTIVE_GEMINI_MODELS", JSON.stringify(modelNames), 21600); return modelNames; } } } } catch (err) {} return ["models/gemini-2.5-flash", "models/gemini-2.0-flash", "models/gemini-1.5-flash"]; } function fetchGeminiWithDynamicFallback(promptText, maxTokens) { var props = PropertiesService.getScriptProperties(); var apiKeys = []; if (props.getProperty("GEMINI_API_KEY")) apiKeys.push(props.getProperty("GEMINI_API_KEY")); if (props.getProperty("GEMINI_API_KEY_1")) apiKeys.push(props.getProperty("GEMINI_API_KEY_1")); if (props.getProperty("GEMINI_API_KEY_2")) apiKeys.push(props.getProperty("GEMINI_API_KEY_2")); if (apiKeys.length === 0) { throw new Error("GEMINI_API_KEY がスクリプトプロパティに設定されていません。"); } var payload = { contents: [{ parts: [{ text: promptText }] }], generationConfig: { temperature: 0.2, maxOutputTokens: maxTokens || 1500 } }; var lastError = ""; for (var k = 0; k < apiKeys.length; k++) { var currentKey = apiKeys[k]; var candidateModels = getActiveGeminiModels(currentKey); for (var m = 0; m < candidateModels.length; m++) { var currentModel = candidateModels[m]; var url = "https://generativelanguage.googleapis.com/v1beta/" + currentModel + ":generateContent?key=" + currentKey; var options = { method: "post", contentType: "application/json", payload: JSON.stringify(payload), muteHttpExceptions: true }; try { var response = UrlFetchApp.fetch(url, options); var responseCode = response.getResponseCode(); var json = JSON.parse(response.getContentText()); if (responseCode === 200 && json.candidates && json.candidates[0] && json.candidates[0].content) { return json.candidates[0].content.parts[0].text; } if (responseCode === 429 || (json.error && json.error.message && json.error.message.indexOf("RESOURCE_EXHAUSTED") !== -1)) { lastError = "モデル [" + currentModel + "] の上限到達。"; Utilities.sleep(1000); continue; } if (json.error) lastError = "API Error (" + responseCode + "): " + json.error.message; } catch (err) { lastError = err.toString(); } } } throw new Error("⚠️ AIモデルへのアクセスが集中しています。時間をおいて再送信してください。\n詳細: " + lastError); } function saveLogToSpreadsheet(data) { var lock = LockService.getScriptLock(); try { lock.waitLock(30000); var spreadsheetId = PropertiesService.getScriptProperties().getProperty("SPREADSHEET_ID"); var sheet = spreadsheetId ? SpreadsheetApp.openById(spreadsheetId).getActiveSheet() : SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); if (sheet.getLastRow() === 0) { sheet.appendRow([ "日時", "メールアドレス", "学年", "ホーム(組)", "出席番号", "氏名", "トピック", "字数制限", "初版英文", "修正版1", "修正版2(最終)", "初版点数", "改善点数", "合計点数", "対話回答", "最終レポート" ]); } var initNum = parseInt((data.initialScore || "").replace(/[^0-9]/g, ""), 10) || 0; var impNum = parseInt((data.improvementScore || "").replace(/[^0-9]/g, ""), 10) || 0; sheet.appendRow([ new Date(), data.userEmail || "", data.userGrade || "", data.userClass || "", data.userNumber || "", data.userName || "", data.topic || "", data.wordLimit || "", data.originalEssay || "", data.revised1 || "", data.revised2 || "", initNum, impNum, initNum + impNum, JSON.stringify(data.interviewAnswers || []), data.finalReportText || "" ]); } finally { lock.releaseLock(); } } function getHistoryFromSpreadsheet(userEmail) { var spreadsheetId = PropertiesService.getScriptProperties().getProperty("SPREADSHEET_ID"); var sheet = spreadsheetId ? SpreadsheetApp.openById(spreadsheetId).getActiveSheet() : SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return []; var logs = []; for (var i = rows.length - 1; i >= 1; i--) { var row = rows[i]; var emailInRow = row[1] ? row[1].toString().trim() : ""; if (!userEmail || emailInRow.toLowerCase() === userEmail.trim().toLowerCase()) { logs.push({ timestamp: row[0] ? Utilities.formatDate(new Date(row[0]), "JST", "yyyy/MM/dd HH:mm") : "", userEmail: emailInRow, userGrade: row[2] || "", userClass: row[3] || "", userNumber: row[4] || "", userName: row[5] || "", topic: row[6] || "", wordLimit: row[7] || "", originalEssay: row[8] || "", revised1: row[9] || "", revised2: row[10] || "", initScore: row[11] || 0, impScore: row[12] || 0, totalScore: row[13] || 0, answers: row[14] || "[]", finalReport: row[15] || "" }); } } return logs; } function createJsonResponse(data) { return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON); }
